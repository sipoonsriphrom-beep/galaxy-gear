from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from datetime import datetime
import sqlite3

app = FastAPI(title="Galaxy Gear API")

# Model สำหรับรับข้อมูลชำระเงิน
class CheckoutRequest(BaseModel):
    username: str
    items: list
    total_price: float

@app.post("/api/checkout")
def process_checkout(data: CheckoutRequest):
    now = datetime.now()  # บันทึกเวลาฝั่ง Python Backend
    
    conn = sqlite3.connect("galaxy_shop.db")
    cursor = conn.cursor()
    
    # ตรวจสอบเครดิตผู้ใช้
    cursor.execute("SELECT credit FROM users WHERE username = ?", (data.username,))
    user = cursor.fetchone()
    
    if not user or user[0] < data.total_price:
        conn.close()
        raise HTTPException(status_code=400, detail="เครดิตไม่เพียงพอ")
        
    # หักเครดิต & บันทึกคำสั่งซื้อลง SQLite
    new_credit = user[0] - data.total_price
    cursor.execute("UPDATE users SET credit = ? WHERE username = ?", (new_credit, data.username))
    cursor.execute("INSERT INTO orders (username, total, created_at) VALUES (?, ?, ?)", 
                   (data.username, data.total_price, now.strftime("%Y-%m-%d %H:%M:%S")))
    
    conn.commit()
    conn.close()
    
    return {
        "status": "success",
        "timestamp": now.strftime("%Y-%m-%d %H:%M:%S"),
        "remaining_credit": new_credit
    }
